469: /// native scopes. Barriers are placed before the callback, never between two

471: fn combine_uses(uses: &mut Vec<RangeUse>, start: usize, added: RangeUse) -> Result<()> {
472:     let before = uses.len();
473:     let mut remaining = smallvec::SmallVec::<[NativeRange; 8]>::from_slice(&[added.range]);
474:     for index in start..before {
475:         let previous = uses[index];
476:         if previous.resource != added.resource {
477:             continue;
478:         }
479:         let Some(overlap) = crate::resource::range_state::intersection(previous.range, added.range)
480:         else {
481:             continue;
482:         };
483:         if previous.layout != added.layout {
484:             return Err(Error::invalid_usage(
485:                 "overlapping image declarations in one pass require one native layout",
486:             ));
487:         }
488:         let outside = crate::resource::range_state::difference(previous.range, overlap);
489:         uses[index] = RangeUse {
490:             range: overlap,
491:             scope: previous.scope.union(added.scope),
492:             ..previous
493:         };
494:         uses.extend(
495:             outside
496:                 .into_iter()
497:                 .map(|range| RangeUse { range, ..previous }),
498:         );
499:         let mut next = smallvec::SmallVec::<[NativeRange; 8]>::new();
500:         for range in remaining {
501:             next.extend(crate::resource::range_state::difference(
502:                 range,
503:                 previous.range,
504:             ));
505:         }
506:         remaining = next;
507:     }
508:     uses.extend(
509:         remaining
510:             .into_iter()
511:             .map(|range| RangeUse { range, ..added }),
512:     );
